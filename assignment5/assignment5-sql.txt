SELECT 
SUM(products.price * line_items.quantity) AS total_price, 
orders.order_id 
FROM orders 
JOIN line_items 
ON orders.order_id = line_items.order_id 
JOIN products ON line_items.product_id = products.product_id
GROUP BY orders.order_id 
ORDER BY orders.order_id 
LIMIT 5;

SELECT 
customers.customer_name,
ROUND(AVG(total_price)::numeric, 2) AS average_order_price, 
customers.customer_id 
FROM customers 
JOIN (
  SELECT SUM(products.price*line_items.quantity) AS total_price, orders.order_id, customers.customer_id
  FROM customers
  JOIN orders ON orders.customer_id = customers.customer_id 
  JOIN line_items ON orders.order_id = line_items.order_id 
  JOIN products ON line_items.product_id = products.product_id 
  GROUP BY orders.order_id, customers.customer_id
) AS subT 
ON customers.customer_id = subT.customer_id
GROUP BY customers.customer_name, customers.customer_id
ORDER BY customers.customer_name;

SELECT 
last_name, 
CONCAT(last_name, ' ', first_name) AS full_name,
COUNT(orders.order_id) AS order_count 
FROM employees 
JOIN orders 
ON employees.employee_id = orders.employee_id 
GROUP BY employees.employee_id 
HAVING COUNT(orders.order_id) > 5 
ORDER BY full_name ASC;
